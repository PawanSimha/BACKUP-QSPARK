DECLARE
    l_body           CLOB;
    l_booking_id     NUMBER;
    l_user_id        NUMBER;
    l_slot_id        NUMBER;
    l_vtype_id       NUMBER;
    l_booking_status VARCHAR2(20);
    l_base_amount    NUMBER(8, 2);
    l_scheduled_end  TIMESTAMP;
    l_exit_time      TIMESTAMP := CURRENT_TIMESTAMP;
    
    l_sim_late_hours NUMBER := NULL;
    l_overstay_hours NUMBER(5, 2) := 0.00;
    l_penalty_rate   NUMBER(6, 2) := 0.00;
    l_penalty_amount NUMBER(8, 2) := 0.00;
    l_final_total    NUMBER(8, 2) := 0.00;
    
    l_wallet_bal     NUMBER(10, 2);
    l_new_wallet_bal NUMBER(10, 2);
    l_step           VARCHAR2(50) := 'INIT';
BEGIN
    l_step := 'READ_PARAMS';
    l_booking_id := TO_NUMBER(:id);
    l_body       := :body_text;
 
    -- 1. Read optional simulation parameter from body
    IF l_body IS NOT NULL AND DBMS_LOB.GETLENGTH(l_body) > 0 THEN
        BEGIN
            APEX_JSON.PARSE(l_body);
            l_sim_late_hours := APEX_JSON.GET_NUMBER(p_path => 'simulated_late_hours');
        EXCEPTION
            WHEN OTHERS THEN
                l_sim_late_hours := NULL;
        END;
    END IF;
 
    -- 2. Lock and fetch booking details
    l_step := 'LOCK_BOOKING';
    BEGIN
        SELECT user_id, slot_id, vehicle_type_id, status, scheduled_end, base_amount
          INTO l_user_id, l_slot_id, l_vtype_id, l_booking_status, l_scheduled_end, l_base_amount
          FROM qspark_bookings
         WHERE booking_id = l_booking_id
           FOR UPDATE;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            :status_code := 404;
            HTP.P('{"error": "Booking ID ' || l_booking_id || ' not found"}');
            RETURN;
    END;
 
    IF l_booking_status NOT IN ('CONFIRMED', 'CHECKED_IN') THEN
        ROLLBACK;
        :status_code := 409;
        HTP.P('{"error": "Booking is already ' || l_booking_status || '"}');
        RETURN;
    END IF;
 
    -- 3. Fetch penalty rate for vehicle type
    l_step := 'FETCH_PENALTY_RATE';
    SELECT overstay_penalty_hour
      INTO l_penalty_rate
      FROM qspark_vehicle_types
     WHERE vehicle_type_id = l_vtype_id;
 
    -- 4. Calculate overstay duration and penalty
    l_step := 'CALCULATE_PENALTY';
    IF l_sim_late_hours IS NOT NULL THEN
        -- Simulator mode: caller provided explicit lateness
        IF l_sim_late_hours > (10.0 / 60.0) THEN
            l_overstay_hours := ROUND(l_sim_late_hours, 2);
            l_penalty_amount := ROUND(l_overstay_hours * l_penalty_rate, 2);
        ELSE
            l_overstay_hours := 0.00;
            l_penalty_amount := 0.00;
        END IF;
    ELSE
        -- Real-time mode: calculate difference from scheduled_end
        IF l_exit_time > (l_scheduled_end + NUMTODSINTERVAL(10, 'MINUTE')) THEN
            l_overstay_hours := ROUND(
                EXTRACT(DAY    FROM (l_exit_time - l_scheduled_end)) * 24 +
                EXTRACT(HOUR   FROM (l_exit_time - l_scheduled_end)) +
                EXTRACT(MINUTE FROM (l_exit_time - l_scheduled_end)) / 60.0,
                2
            );
            l_penalty_amount := ROUND(l_overstay_hours * l_penalty_rate, 2);
        ELSE
            l_overstay_hours := 0.00;
            l_penalty_amount := 0.00;
        END IF;
    END IF;
 
    l_final_total := l_base_amount + l_penalty_amount;
 
    -- 5. Lock user wallet and verify balance if penalty applies
    l_step := 'PROCESS_WALLET';
    SELECT wallet_balance
      INTO l_wallet_bal
      FROM qspark_users
     WHERE user_id = l_user_id
       FOR UPDATE;
 
    IF l_penalty_amount > 0 THEN
        IF l_wallet_bal < l_penalty_amount THEN
            ROLLBACK;
            :status_code := 402;
            HTP.P('{"error": "Insufficient wallet balance to pay overstay penalty of ₹' || TO_CHAR(l_penalty_amount, 'FM9990.00') || '"}');
            RETURN;
        END IF;
 
        l_new_wallet_bal := l_wallet_bal - l_penalty_amount;
 
        UPDATE qspark_users
           SET wallet_balance = l_new_wallet_bal
         WHERE user_id = l_user_id;
 
        INSERT INTO qspark_wallet_transactions (
            user_id,
            booking_id,
            txn_type,
            previous_balance,
            amount,
            new_balance,
            status
        ) VALUES (
            l_user_id,
            l_booking_id,
            'OVERSTAY_DEBIT',
            l_wallet_bal,
            l_penalty_amount,
            l_new_wallet_bal,
            'COMPLETED'
        );
    ELSE
        l_new_wallet_bal := l_wallet_bal;
    END IF;
 
    -- 6. Free the parking slot
    l_step := 'FREE_SLOT';
    UPDATE qspark_parking_slots
       SET status = 'AVAILABLE',
           last_updated = CURRENT_TIMESTAMP
     WHERE slot_id = l_slot_id;
 
    -- 7. Complete the booking record
    l_step := 'UPDATE_BOOKING';
    UPDATE qspark_bookings
       SET actual_exit_time = l_exit_time,
           overstay_hours   = l_overstay_hours,
           overstay_penalty = l_penalty_amount,
           total_amount     = l_final_total,
           status           = 'COMPLETED'
     WHERE booking_id = l_booking_id;
 
    -- 8. Audit log entry
    l_step := 'INSERT_AUDIT';
    INSERT INTO qspark_audit_logs (
        changed_by_user_id,
        slot_id,
        booking_id,
        event_type,
        old_state,
        new_state,
        log_details
    ) VALUES (
        l_user_id,
        l_slot_id,
        l_booking_id,
        'CHECKOUT_COMPLETED',
        'RESERVED',
        'AVAILABLE',
        'Checkout completed. Penalty: ₹' || TO_CHAR(l_penalty_amount, 'FM9990.00') || '. Total: ₹' || TO_CHAR(l_final_total, 'FM9990.00')
    );
 
    COMMIT;
 
    -- 9. Success response
    :status_code := 200;
    HTP.P('{');
    HTP.P('"booking_id": ' || l_booking_id || ',');
    HTP.P('"overstay_hours": ' || TO_CHAR(l_overstay_hours, 'FM9990.00') || ',');
    HTP.P('"penalty_charged": ' || TO_CHAR(l_penalty_amount, 'FM9990.00') || ',');
    HTP.P('"final_total_amount": ' || TO_CHAR(l_final_total, 'FM9990.00') || ',');
    HTP.P('"new_wallet_balance": ' || TO_CHAR(l_new_wallet_bal, 'FM9990.00') || ',');
    HTP.P('"status": "COMPLETED"');
    HTP.P('}');
 
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        :status_code := 500;
        HTP.P('{');
        HTP.P('"failed_at_step": "' || l_step || '",');
        HTP.P('"error_code": ' || SQLCODE || ',');
        HTP.P('"error_message": ' || APEX_JSON.STRINGIFY(SQLERRM));
        HTP.P('}');
END;
