DECLARE
    l_booking_id     NUMBER;
    l_user_id        NUMBER;
    l_slot_id        NUMBER;
    l_slot_number    VARCHAR2(15);
    l_booking_status VARCHAR2(20);
    l_base_amount    NUMBER(8, 2);
    l_wallet_bal     NUMBER(10, 2);
    l_new_wallet_bal NUMBER(10, 2);
    l_step           VARCHAR2(50) := 'INIT';
BEGIN
    l_step := 'READ_PARAMS';
    l_booking_id := TO_NUMBER(:id);
 
    -- 1. Lock and validate booking
    l_step := 'LOCK_BOOKING';
    BEGIN
        SELECT b.user_id, b.slot_id, b.status, b.base_amount, s.slot_number
          INTO l_user_id, l_slot_id, l_booking_status, l_base_amount, l_slot_number
          FROM qspark_bookings b
          JOIN qspark_parking_slots s ON s.slot_id = b.slot_id
         WHERE b.booking_id = l_booking_id
           FOR UPDATE OF b.status;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            :status_code := 404;
            HTP.P('{"error": "Booking ID ' || l_booking_id || ' not found"}');
            RETURN;
    END;
 
    IF l_booking_status != 'CONFIRMED' THEN
        ROLLBACK;
        :status_code := 409;
        HTP.P('{"error": "Only CONFIRMED bookings can be cancelled. Current status: ' || l_booking_status || '"}');
        RETURN;
    END IF;
 
    -- 2. Lock user wallet and refund base amount
    l_step := 'REFUND_WALLET';
    SELECT wallet_balance
      INTO l_wallet_bal
      FROM qspark_users
     WHERE user_id = l_user_id
       FOR UPDATE;
 
    l_new_wallet_bal := l_wallet_bal + l_base_amount;
 
    UPDATE qspark_users
       SET wallet_balance = l_new_wallet_bal
     WHERE user_id = l_user_id;
 
    -- 3. Free the parking slot
    l_step := 'FREE_SLOT';
    UPDATE qspark_parking_slots
       SET status = 'AVAILABLE',
           last_updated = CURRENT_TIMESTAMP
     WHERE slot_id = l_slot_id;
 
    -- 4. Mark booking CANCELLED
    l_step := 'UPDATE_BOOKING';
    UPDATE qspark_bookings
       SET status = 'CANCELLED'
     WHERE booking_id = l_booking_id;
 
    -- 5. Record refund transaction
    l_step := 'INSERT_LEDGER';
    INSERT INTO qspark_wallet_transactions (
        user_id,
        booking_id,
        txn_type,
        previous_balance,
        amount,
        new_balance,
        status
    ) VALUES (
        l_user_id,
        l_booking_id,
        'CANCEL_REFUND',
        l_wallet_bal,
        l_base_amount,
        l_new_wallet_bal,
        'COMPLETED'
    );
 
    -- 6. Log audit event
    l_step := 'INSERT_AUDIT';
    INSERT INTO qspark_audit_logs (
        changed_by_user_id,
        slot_id,
        booking_id,
        event_type,
        old_state,
        new_state,
        log_details
    ) VALUES (
        l_user_id,
        l_slot_id,
        l_booking_id,
        'BOOKING_CANCELLED',
        'RESERVED',
        'AVAILABLE',
        'Booking cancelled by user. Full refund of ₹' || TO_CHAR(l_base_amount, 'FM9990.00') || ' credited to wallet.'
    );
 
    COMMIT;
 
    -- 7. Response
    :status_code := 200;
    HTP.P('{');
    HTP.P('"booking_id": ' || l_booking_id || ',');
    HTP.P('"refunded_amount": ' || TO_CHAR(l_base_amount, 'FM9990.00') || ',');
    HTP.P('"restored_wallet_balance": ' || TO_CHAR(l_new_wallet_bal, 'FM9990.00') || ',');
    HTP.P('"slot_number": "' || l_slot_number || '",');
    HTP.P('"slot_status": "AVAILABLE",');
    HTP.P('"status": "CANCELLED"');
    HTP.P('}');
 
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        :status_code := 500;
        HTP.P('{');
        HTP.P('"failed_at_step": "' || l_step || '",');
        HTP.P('"error_code": ' || SQLCODE || ',');
        HTP.P('"error_message": ' || APEX_JSON.STRINGIFY(SQLERRM));
        HTP.P('}');
END;
