 CREATE TABLE QSPARK_USERS (
    user_id        NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 1001) PRIMARY KEY,
    username       VARCHAR2(50)  NOT NULL,
    email          VARCHAR2(100) NOT NULL,
    password_hash  VARCHAR2(255) NOT NULL,
    full_name      VARCHAR2(100) NOT NULL,
    role           VARCHAR2(10)  DEFAULT 'USER' NOT NULL,
    wallet_balance NUMBER(10, 2) DEFAULT 500.00 NOT NULL,
    is_active      CHAR(1)       DEFAULT 'Y' NOT NULL,
    created_at     TIMESTAMP     DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT unq_qspark_users_username UNIQUE (username),
    CONSTRAINT unq_qspark_users_email    UNIQUE (email),
    CONSTRAINT chk_qspark_users_role     CHECK (role IN ('ADMIN', 'USER')),
    CONSTRAINT chk_qspark_users_wallet   CHECK (wallet_balance >= 0),
    CONSTRAINT chk_qspark_users_active   CHECK (is_active IN ('Y', 'N'))
);

SELECT * FROM QSPARK_USERS;

CREATE TABLE QSPARK_FLOORS (
    floor_id       NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    floor_code     VARCHAR2(5)  NOT NULL,
    floor_name     VARCHAR2(50) NOT NULL,
    bike_capacity  NUMBER(3)    DEFAULT 40 NOT NULL,
    car_capacity   NUMBER(3)    DEFAULT 20 NOT NULL,
    total_capacity NUMBER(3)    DEFAULT 60 NOT NULL,
    status         VARCHAR2(15) DEFAULT 'OPEN' NOT NULL,
    CONSTRAINT unq_qspark_floors_code   UNIQUE (floor_code),
    CONSTRAINT chk_qspark_floors_code   CHECK (floor_code IN ('B1', 'B2', 'B3', 'B4')),
    CONSTRAINT chk_qspark_floors_bike   CHECK (bike_capacity = 40),
    CONSTRAINT chk_qspark_floors_car    CHECK (car_capacity = 20),
    CONSTRAINT chk_qspark_floors_total  CHECK (total_capacity = 60),
    CONSTRAINT chk_qspark_floors_status CHECK (status IN ('OPEN', 'CLOSED', 'MAINTENANCE'))
);

 SELECT * FROM QSPARK_FLOORS;

CREATE TABLE QSPARK_VEHICLE_TYPES (
    vehicle_type_id       NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    type_code             VARCHAR2(15) NOT NULL,
    display_name          VARCHAR2(50) NOT NULL,
    hourly_rate           NUMBER(6, 2) NOT NULL,
    overstay_penalty_hour NUMBER(6, 2) DEFAULT 10.00 NOT NULL,
    total_slots_per_floor NUMBER(3)    NOT NULL,
    CONSTRAINT unq_qspark_vtypes_code    UNIQUE (type_code),
    CONSTRAINT chk_qspark_vtypes_code    CHECK (type_code IN ('BIKE', 'CAR')),
    CONSTRAINT chk_qspark_vtypes_rate    CHECK (hourly_rate IN (5.00, 10.00)),
    CONSTRAINT chk_qspark_vtypes_penalty CHECK (overstay_penalty_hour >= 0),
    CONSTRAINT chk_qspark_vtypes_slots   CHECK (total_slots_per_floor IN (40, 20))
);

 SELECT * FROM QSPARK_VEHICLE_TYPES; 

--  ============================================================================

CREATE TABLE QSPARK_PARKING_SLOTS (
    slot_id         NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    floor_id        NUMBER       NOT NULL,
    vehicle_type_id NUMBER       NOT NULL,
    slot_number     VARCHAR2(15) NOT NULL,
    status          VARCHAR2(20) DEFAULT 'AVAILABLE' NOT NULL,
    last_updated    TIMESTAMP    DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT fk_qspark_slots_floor  FOREIGN KEY (floor_id) REFERENCES QSPARK_FLOORS(floor_id) ON DELETE CASCADE,
    CONSTRAINT fk_qspark_slots_vtype  FOREIGN KEY (vehicle_type_id) REFERENCES QSPARK_VEHICLE_TYPES(vehicle_type_id),
    CONSTRAINT unq_qspark_slots_num   UNIQUE (slot_number),
    CONSTRAINT unq_qspark_floor_slot  UNIQUE (floor_id, slot_number),
    CONSTRAINT chk_qspark_slot_status CHECK (status IN ('AVAILABLE', 'RESERVED', 'OCCUPIED', 'MAINTENANCE'))
);

SELECT * FROM QSPARK_PARKING_SLOTS;

CREATE TABLE QSPARK_BOOKINGS (
    booking_id          NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 10001) PRIMARY KEY,
    booking_code        VARCHAR2(25) NOT NULL,
    user_id             NUMBER       NOT NULL,
    slot_id             NUMBER       NOT NULL,
    vehicle_type_id     NUMBER       NOT NULL,
    vehicle_number      VARCHAR2(20) NOT NULL,
    scheduled_start     TIMESTAMP    NOT NULL,
    scheduled_end       TIMESTAMP    NOT NULL,
    actual_entry_time   TIMESTAMP,
    actual_exit_time    TIMESTAMP,
    duration_hours      NUMBER(4, 1) NOT NULL,
    hourly_rate_applied NUMBER(6, 2) NOT NULL,
    base_amount         NUMBER(8, 2) NOT NULL,
    overstay_hours      NUMBER(5, 2) DEFAULT 0.00 NOT NULL,
    overstay_penalty    NUMBER(8, 2) DEFAULT 0.00 NOT NULL,
    total_amount        NUMBER(8, 2) NOT NULL,
    status              VARCHAR2(20) DEFAULT 'CONFIRMED' NOT NULL,
    created_at          TIMESTAMP    DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT unq_qspark_booking_code   UNIQUE (booking_code),
    CONSTRAINT fk_qspark_bookings_user   FOREIGN KEY (user_id) REFERENCES QSPARK_USERS(user_id),
    CONSTRAINT fk_qspark_bookings_slot   FOREIGN KEY (slot_id) REFERENCES QSPARK_PARKING_SLOTS(slot_id),
    CONSTRAINT fk_qspark_bookings_vtype  FOREIGN KEY (vehicle_type_id) REFERENCES QSPARK_VEHICLE_TYPES(vehicle_type_id),
    CONSTRAINT chk_qspark_booking_dur    CHECK (duration_hours >= 1.0),
    CONSTRAINT chk_qspark_booking_rate   CHECK (hourly_rate_applied IN (5.00, 10.00)),
    CONSTRAINT chk_qspark_booking_base   CHECK (base_amount >= 5.00),
    CONSTRAINT chk_qspark_booking_ovhrs  CHECK (overstay_hours >= 0),
    CONSTRAINT chk_qspark_booking_ovpen  CHECK (overstay_penalty >= 0),
    CONSTRAINT chk_qspark_booking_total  CHECK (total_amount >= 0),
    CONSTRAINT chk_qspark_booking_status CHECK (status IN ('CONFIRMED', 'CHECKED_IN', 'COMPLETED', 'CANCELLED', 'OVERSTAY'))
);

CREATE TABLE QSPARK_WALLET_TRANSACTIONS (
    txn_id           NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 20001) PRIMARY KEY,
    user_id          NUMBER        NOT NULL,
    booking_id       NUMBER        NOT NULL,
    txn_type         VARCHAR2(25)  NOT NULL,
    previous_balance NUMBER(10, 2) NOT NULL,
    amount           NUMBER(8, 2)  NOT NULL,
    new_balance      NUMBER(10, 2) NOT NULL,
    status           VARCHAR2(15)  DEFAULT 'COMPLETED' NOT NULL,
    processed_at     TIMESTAMP     DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT fk_qspark_wallet_user    FOREIGN KEY (user_id) REFERENCES QSPARK_USERS(user_id),
    CONSTRAINT fk_qspark_wallet_booking FOREIGN KEY (booking_id) REFERENCES QSPARK_BOOKINGS(booking_id) ON DELETE CASCADE,
    CONSTRAINT chk_qspark_wallet_type   CHECK (txn_type IN ('BOOKING_DEBIT', 'OVERSTAY_DEBIT', 'CANCEL_REFUND')),
    CONSTRAINT chk_qspark_wallet_amt    CHECK (amount >= 0),
    CONSTRAINT chk_qspark_wallet_newbal CHECK (new_balance >= 0),
    CONSTRAINT chk_qspark_wallet_status CHECK (status IN ('COMPLETED', 'REVERSED'))
);

SELECT * FROM QSPARK_WALLET_TRANSACTIONS;

CREATE TABLE QSPARK_AUDIT_LOGS (
    log_id             NUMBER GENERATED BY DEFAULT AS IDENTITY (START WITH 1) PRIMARY KEY,
    changed_by_user_id NUMBER,
    slot_id            NUMBER,
    booking_id         NUMBER,
    event_type         VARCHAR2(30)  NOT NULL,
    old_state          VARCHAR2(50),
    new_state          VARCHAR2(50),
    log_details        VARCHAR2(500),
    logged_at          TIMESTAMP     DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT fk_qspark_audit_user    FOREIGN KEY (changed_by_user_id) REFERENCES QSPARK_USERS(user_id),
    CONSTRAINT fk_qspark_audit_slot    FOREIGN KEY (slot_id) REFERENCES QSPARK_PARKING_SLOTS(slot_id) ON DELETE CASCADE,
    CONSTRAINT fk_qspark_audit_booking FOREIGN KEY (booking_id) REFERENCES QSPARK_BOOKINGS(booking_id) ON DELETE CASCADE,
    CONSTRAINT chk_qspark_audit_event  CHECK (event_type IN ('SLOT_STATUS_CHANGE', 'BOOKING_CREATED', 'BOOKING_CANCELLED', 'CHECKOUT_COMPLETED', 'ADMIN_OVERRIDE'))
);

SELECT * FROM QSPARK_AUDIT_LOGS;
