SELECT
    s.slot_id,
    f.floor_code,
    s.slot_number,
    vt.type_code AS vehicle_type,
    vt.hourly_rate,
    s.status
FROM QSPARK_PARKING_SLOTS s
JOIN QSPARK_FLOORS f
    ON f.floor_id = s.floor_id
JOIN QSPARK_VEHICLE_TYPES vt
    ON vt.vehicle_type_id = s.vehicle_type_id
ORDER BY
    s.slot_id
