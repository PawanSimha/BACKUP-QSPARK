DECLARE
    l_body           CLOB;
    l_user_id        NUMBER;
    l_slot_id        NUMBER;
    l_vehicle_number VARCHAR2(20);
    l_duration_hours NUMBER(4, 1);
    
    l_slot_status    VARCHAR2(20);
    l_slot_number    VARCHAR2(15);
    l_vtype_id       NUMBER;
    l_hourly_rate    NUMBER(6, 2);
    l_base_amount    NUMBER(8, 2);
    
    l_wallet_bal     NUMBER(10, 2);
    l_new_wallet_bal NUMBER(10, 2);
    
    l_booking_id     NUMBER;
    l_booking_code   VARCHAR2(25);
    l_start_time     TIMESTAMP := CURRENT_TIMESTAMP;
    l_end_time       TIMESTAMP;
    l_step           VARCHAR2(50) := 'INIT';
BEGIN
    l_step := 'READ_PARAMS';
    l_body := :body_text;
 
    IF l_body IS NULL OR DBMS_LOB.GETLENGTH(l_body) = 0 THEN
        :status_code := 400;
        HTP.P('{"error": "Request body cannot be empty"}');
        RETURN;
    END IF;
 
    l_step := 'PARSE_JSON';
    APEX_JSON.PARSE(l_body);
    l_user_id        := APEX_JSON.GET_NUMBER(p_path => 'user_id');
    l_slot_id        := APEX_JSON.GET_NUMBER(p_path => 'slot_id');
    l_vehicle_number := UPPER(TRIM(APEX_JSON.GET_VARCHAR2(p_path => 'vehicle_number')));
    l_duration_hours := APEX_JSON.GET_NUMBER(p_path => 'duration_hours');
 
    IF l_user_id IS NULL OR l_slot_id IS NULL OR l_vehicle_number IS NULL OR NVL(l_duration_hours, 0) < 1.0 THEN
        :status_code := 400;
        HTP.P('{"error": "Missing or invalid required fields (user_id, slot_id, vehicle_number, duration_hours >= 1.0)"}');
        RETURN;
    END IF;
 
    l_step := 'LOCK_SLOT';
    BEGIN
        SELECT status, slot_number, vehicle_type_id
          INTO l_slot_status, l_slot_number, l_vtype_id
          FROM qspark_parking_slots
         WHERE slot_id = l_slot_id
           FOR UPDATE;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            :status_code := 404;
            HTP.P('{"error": "Slot ID ' || l_slot_id || ' not found"}');
            RETURN;
    END;
 
    IF l_slot_status != 'AVAILABLE' THEN
        ROLLBACK;
        :status_code := 409;
        HTP.P('{"error": "Slot is already reserved or occupied"}');
        RETURN;
    END IF;
 
    l_step := 'FETCH_RATE';
    SELECT hourly_rate
      INTO l_hourly_rate
      FROM qspark_vehicle_types
     WHERE vehicle_type_id = l_vtype_id;
 
    l_base_amount := l_duration_hours * l_hourly_rate;
    l_end_time    := l_start_time + NUMTODSINTERVAL(l_duration_hours, 'HOUR');
 
    l_step := 'LOCK_WALLET';
    BEGIN
        SELECT wallet_balance
          INTO l_wallet_bal
          FROM qspark_users
         WHERE user_id = l_user_id
           FOR UPDATE;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            ROLLBACK;
            :status_code := 404;
            HTP.P('{"error": "User ID ' || l_user_id || ' not found"}');
            RETURN;
    END;
 
    IF l_wallet_bal < l_base_amount THEN
        ROLLBACK;
        :status_code := 402;
        HTP.P('{"error": "Insufficient Metro Card wallet balance"}');
        RETURN;
    END IF;
 
    l_new_wallet_bal := l_wallet_bal - l_base_amount;
 
    l_step := 'DEBIT_WALLET';
    UPDATE qspark_users
       SET wallet_balance = l_new_wallet_bal
     WHERE user_id = l_user_id;
 
    l_step := 'UPDATE_SLOT_STATUS';
    UPDATE qspark_parking_slots
       SET status = 'RESERVED',
           last_updated = CURRENT_TIMESTAMP
     WHERE slot_id = l_slot_id;
 
    l_step := 'INSERT_BOOKING';
    l_booking_code := 'QSP-' || TO_CHAR(CURRENT_TIMESTAMP, 'MMDD') || '-' || LPAD(TO_CHAR(MOD(ABS(DBMS_RANDOM.RANDOM), 9000) + 1000), 4, '0');
 
    INSERT INTO qspark_bookings (
        booking_code,
        user_id,
        slot_id,
        vehicle_type_id,
        vehicle_number,
        scheduled_start,
        scheduled_end,
        duration_hours,
        hourly_rate_applied,
        base_amount,
        overstay_hours,
        overstay_penalty,
        total_amount,
        status
    ) VALUES (
        l_booking_code,
        l_user_id,
        l_slot_id,
        l_vtype_id,
        l_vehicle_number,
        l_start_time,
        l_end_time,
        l_duration_hours,
        l_hourly_rate,
        l_base_amount,
        0.00,
        0.00,
        l_base_amount,
        'CONFIRMED'
    ) RETURNING booking_id INTO l_booking_id;
 
    l_step := 'INSERT_LEDGER';
    INSERT INTO qspark_wallet_transactions (
        user_id,
        booking_id,
        txn_type,
        previous_balance,
        amount,
        new_balance,
        status
    ) VALUES (
        l_user_id,
        l_booking_id,
        'BOOKING_DEBIT',
        l_wallet_bal,
        l_base_amount,
        l_new_wallet_bal,
        'COMPLETED'
    );
 
    l_step := 'INSERT_AUDIT';
    INSERT INTO qspark_audit_logs (
        changed_by_user_id,
        slot_id,
        booking_id,
        event_type,
        old_state,
        new_state,
        log_details
    ) VALUES (
        l_user_id,
        l_slot_id,
        l_booking_id,
        'BOOKING_CREATED',
        'AVAILABLE',
        'RESERVED',
        'Reservation confirmed with wallet debit of ₹' || TO_CHAR(l_base_amount, 'FM9990.00')
    );
 
    l_step := 'COMMIT';
    COMMIT;
 
    :status_code := 201;
    HTP.P('{');
    HTP.P('"booking_id": ' || l_booking_id || ',');
    HTP.P('"booking_code": "' || l_booking_code || '",');
    HTP.P('"slot_number": "' || l_slot_number || '",');
    HTP.P('"base_amount": ' || TO_CHAR(l_base_amount, 'FM9990.00') || ',');
    HTP.P('"previous_wallet_balance": ' || TO_CHAR(l_wallet_bal, 'FM9990.00') || ',');
    HTP.P('"new_wallet_balance": ' || TO_CHAR(l_new_wallet_bal, 'FM9990.00') || ',');
    HTP.P('"status": "CONFIRMED"');
    HTP.P('}');
 
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        :status_code := 500;
        HTP.P('{');
        HTP.P('"failed_at_step": "' || l_step || '",');
        HTP.P('"error_code": ' || SQLCODE || ',');
        HTP.P('"error_message": ' || APEX_JSON.STRINGIFY(SQLERRM));
        HTP.P('}');
END;
